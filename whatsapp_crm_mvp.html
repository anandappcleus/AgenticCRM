🎯 What You're Building

A SaaS platform that plugs into a local business's WhatsApp number and uses AI agents to automatically reply to customers, follow up on cold leads, suggest products, and give the business owner a live CRM dashboard — all without them writing a single message.

👤 Target Customer
  • Clothing stores, salons, restaurants in Asansol/Durgapur
  • Coaching centers, medical clinics, hardware shops
  • Owner gets 50–150 WhatsApp messages/day
  • Replies manually, misses nights/weekends
  • Has no budget for a full-time assistant
  • Pays ₹1,000–₹3,000/month if it saves time
✨ Core MVP Features (30-day scope)

💡 Philosophy: Keep It Boring and Shippable

The biggest mistake is over-engineering. For your first 5 clients, you don't need microservices, Kubernetes, or a multi-model ensemble. You need a working webhook, one GPT-4o call per message, and a simple database. Build that first. Complexity can come later — after you have paying customers.

📐 Design Principle: One Business = One Context

Each business that subscribes gets their own isolated context: their own product catalog in ChromaDB, their own conversation table partition, and their own system prompt. This way one client's data never leaks into another's.

⚠️ Multi-Tenancy from Day 1

Even with 1 client, design every DB table with a business_id foreign key. This makes adding client #2 trivial and prevents painful refactors later.

from fastapi import APIRouter, Request, BackgroundTasks, HTTPException, Query
from app.config import settings
from app.services.whatsapp import parse_incoming_message
from app.services.pipeline import process_message
import logging

router = APIRouter()
logger = logging.getLogger(__name__)

# ── GET: Meta webhook verification ──────────────────────────
@router.get("/")
async def verify_webhook(
    hub_mode: str = Query(None, alias="hub.mode"),
    hub_challenge: str = Query(None, alias="hub.challenge"),
    hub_verify_token: str = Query(None, alias="hub.verify_token"),
):
    if hub_mode == "subscribe" and hub_verify_token == settings.WHATSAPP_VERIFY_TOKEN:
        return int(hub_challenge)
    raise HTTPException(status_code=403, detail="Invalid verify token")

# ── POST: Incoming WhatsApp messages ────────────────────────
@router.post("/")
async def receive_message(request: Request, bg: BackgroundTasks):
    payload = await request.json()

    try:
        # Parse Meta webhook format
        entry = payload.get("entry", [{}])[0]
        changes = entry.get("changes", [{}])[0]
        value = changes.get("value", {})

        # Skip if not a message event (e.g. status updates)
        if "messages" not in value:
            return {"status": "ok", "skipped": True}

        msg_data = parse_incoming_message(value)

        # Process in background — return 200 immediately to Meta
        bg.add_task(process_message, msg_data)

    except Exception as e:
        logger.error(f"Webhook error: {e}", exc_info=True)

    # Always return 200 to Meta (prevents retries)
    return {"status": "ok"}

app/services/pipeline.py — Core Message Pipeline
python
from app.services.whatsapp import IncomingMessage, send_whatsapp_message
from app.agents.reply_agent import ReplyAgent
from app.agents.sales_agent import SalesAgent
from app.agents.followup_agent import schedule_followup, cancel_followup
from app.models.customer import get_or_create_customer
from app.models.conversation import get_conversation_history, save_message
from app.models.lead import upsert_lead
from app.utils.intent import classify_intent
import logging

logger = logging.getLogger(__name__)

async def process_message(msg: IncomingMessage):
    """
    Main pipeline: WhatsApp message → AI response.
    Runs as a FastAPI background task.
    """
    try:
        # 1. Upsert customer record
        customer = await get_or_create_customer(
            phone=msg.phone, name=msg.name
        )

        # 2. Save incoming message to DB
        await save_message(
            customer_id=customer.id,
            role="user",
            content=msg.text,
            timestamp=msg.timestamp
        )

        # 3. Cancel any pending follow-up (they replied!)
        await cancel_followup(customer.id)

        # 4. Classify intent (quick, no LLM call)
        intent = classify_intent(msg.text)
        logger.info(f"Customer {msg.phone} | Intent: {intent}")

        # 5. Build conversation history (last 10 messages)
        history = await get_conversation_history(customer.id, limit=10)

        # 6. Run Reply Agent → generates response
        reply_agent = ReplyAgent(business_id=customer.business_id)
        response_text = await reply_agent.run(
            customer_message=msg.text,
            history=history,
            customer_name=customer.name,
            intent=intent
        )

        # 7. Sales Agent: append offer if purchase intent
        if intent in ["purchase", "price_inquiry"]:
            sales_agent = SalesAgent(business_id=customer.business_id)
            offer = await sales_agent.get_offer(msg.text)
            if offer:
                response_text += f"\n\n{offer}"

        # 8. Send reply via WhatsApp
        await send_whatsapp_message(msg.phone, response_text)

        # 9. Save AI response to DB
        await save_message(
            customer_id=customer.id,
            role="assistant",
            content=response_text
        )

        # 10. Update lead score + schedule follow-up
        await upsert_lead(customer.id, intent=intent)
        await schedule_followup(customer.id, msg.phone, intent=intent)

    except Exception as e:
        logger.error(f"Pipeline error for {msg.phone}: {e}", exc_info=True)

from openai import AsyncOpenAI
from app.config import settings
from app.services.rag import BusinessRAG
from typing import List, Dict

client = AsyncOpenAI(api_key=settings.OPENAI_API_KEY)

class ReplyAgent:
    def __init__(self, business_id: str):
        self.business_id = business_id
        self.rag = BusinessRAG(business_id)

    async def run(
        self,
        customer_message: str,
        history: List[Dict],
        customer_name: str,
        intent: str
    ) -> str:
        # Fetch relevant business knowledge via RAG
        context = await self.rag.query(customer_message, top_k=4)

        # Build the system prompt
        system_prompt = self._build_system_prompt(context)

        # Build message chain: system + history + new user message
        messages = [
            {"role": "system", "content": system_prompt},
            *history,  # Last 10 messages from DB
            {"role": "user", "content": customer_message}
        ]

        response = await client.chat.completions.create(
            model=settings.OPENAI_MODEL,
            messages=messages,
            max_tokens=300,
            temperature=0.7,
        )
        return response.choices[0].message.content.strip()

    def _build_system_prompt(self, rag_context: str) -> str:
        return f"""You are a helpful WhatsApp assistant for a local Indian business.

ROLE:
- You reply on behalf of the business owner
- You are friendly, warm, and professional
- Speak in Hinglish (mix of Hindi and English) unless customer uses Bengali

LANGUAGE RULES:
- Default: Hinglish (e.g., "Aapka jacket ₹850 mein milega!")
- If customer writes in Hindi → reply in Hindi
- If customer writes in English → reply in English
- Never use formal/robotic language. Sound like a real shopkeeper.
- Use "Aap" for respect, not "Tum"
- End responses with a helpful question or call to action

TONE EXAMPLES:
✅ Good: "Haan ji! Yeh wali jacket abhi available hai, ₹850 mein. Aapko kaisa size chahiye?"
❌ Bad: "Yes, the jacket is available at Rs. 850. Please let me know your size requirements."

BUSINESS KNOWLEDGE (use this to answer questions):
{rag_context}

IMPORTANT RULES:
1. Never make up prices or products not in the knowledge base
2. If you don't know something, say "Main abhi check karke batata hoon"
3. Never promise delivery dates you can't confirm
4. Keep replies SHORT — 2-4 sentences max (WhatsApp readers)
5. If customer seems ready to buy, ask for their address/size/preference to close"""
app/services/rag.py — Business Knowledge RAG Pipeline
python
import chromadb
from openai import AsyncOpenAI
from app.config import settings
from typing import List

client = AsyncOpenAI(api_key=settings.OPENAI_API_KEY)

class BusinessRAG:
    def __init__(self, business_id: str):
        self.business_id = business_id
        # Each business gets its own ChromaDB collection
        self.chroma = chromadb.PersistentClient(path=settings.CHROMA_PATH)
        self.collection = self.chroma.get_or_create_collection(
            name=f"business_{business_id}",
            metadata={"hnsw:space": "cosine"}
        )

    async def query(self, text: str, top_k: int = 4) -> str:
        """Query business knowledge for relevant context."""
        if self.collection.count() == 0:
            return "No business knowledge loaded yet."

        # Get embedding for query
        embed_resp = await client.embeddings.create(
            input=text,
            model=settings.OPENAI_EMBED_MODEL
        )
        query_embedding = embed_resp.data[0].embedding

        # Search ChromaDB
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=min(top_k, self.collection.count())
        )

        # Format results into context string
        docs = results["documents"][0]
        return "\n---\n".join(docs)

    async def ingest_catalog(self, items: List[Dict]):
        """Upload business product catalog / FAQs to ChromaDB."""
        texts, ids, metas = [], [], []

        for i, item in enumerate(items):
            # Format each item as searchable text
            text = f"""Product: {item['name']}
Price: ₹{item['price']}
Description: {item['description']}
Category: {item.get('category', 'General')}
Available: {item.get('available', True)}"""
            texts.append(text)
            ids.append(f"{self.business_id}_{i}")
            metas.append({"type": "product", "name": item["name"]})

        # Embed all texts at once (batch call)
        embed_resp = await client.embeddings.create(
            input=texts, model=settings.OPENAI_EMBED_MODEL
        )
        embeddings = [d.embedding for d in embed_resp.data]

        self.collection.upsert(
            documents=texts, embeddings=embeddings,
            ids=ids, metadatas=metas
        )
        return {"ingested": len(texts)}

async def schedule_followup(customer_id: str, phone: str, intent: str):
    """Schedule a follow-up if customer goes quiet."""
    job_id = f"followup_{customer_id}"

    # Remove existing follow-up if any
    if scheduler.get_job(job_id):
        scheduler.remove_job(job_id)

    # Only follow up on purchase/inquiry intents
    if intent not in ["purchase", "price_inquiry", "general"]:
        return

    run_time = datetime.now() + timedelta(hours=settings.FOLLOWUP_HOURS)
    scheduler.add_job(
        send_followup_message,
        trigger="date",
        run_date=run_time,
        args=[phone, intent],
        id=job_id,
        replace_existing=True
    )

async def cancel_followup(customer_id: str):
    """Cancel follow-up when customer replies (they're active again)."""
    job_id = f"followup_{customer_id}"
    if scheduler.get_job(job_id):
        scheduler.remove_job(job_id)

async def send_followup_message(phone: str, intent: str):
    """Generate and send a personalized follow-up via AI."""
    prompt = f"""Write a short, warm WhatsApp follow-up message in Hinglish.
Context: Customer showed {intent} intent but went quiet.
Rules: Max 2 sentences. Friendly. Not pushy. End with a question.
Example: "Namaste! Kya aap abhi bhi jacket mein interested hain? 😊""""

    resp = await client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}],
        max_tokens=100
    )
    message = resp.choices[0].message.content.strip()
    await send_whatsapp_message(phone, message)

"""
Fast, zero-cost intent classifier using keyword matching.
Use this BEFORE the GPT call to avoid unnecessary API costs.
Handles Hinglish + Hindi + English keywords.
"""

INTENT_KEYWORDS = {
    "purchase": [
        "order", "buy", "khareedna", "chahiye", "lena hai",
        "book", "confirm", "place order", "ready"
    ],
    "price_inquiry": [
        "price", "kitna", "rate", "cost", "daam",
        "rupee", "₹", "discount", "offer", "cheap"
    ],
    "complaint": [
        "problem", "issue", "wrong", "refund", "return",
        "damage", "bura", "complaint", "angry", "worst"
    ],
    "hours": [
        "open", "close", "time", "timing", "kab",
        "khula", "band", "sunday", "holiday"
    ],
    "greeting": [
        "hello", "hi", "namaste", "haan", "hey",
        "good morning", "bhai", "bhaiya"
    ],
}

def classify_intent(text: str) -> str:
    text_lower = text.lower()
    for intent, keywords in INTENT_KEYWORDS.items():
        if any(kw in text_lower for kw in keywords):
            return intent
    return "general"  # default fallback

struct ChatView: View {
    let customer: Customer
    @StateObject private var vm: ChatViewModel
    @State private var replyText = ""
    @State private var isAIPaused = false

    init(customer: Customer) {
        self.customer = customer
        _vm = StateObject(wrappedValue: ChatViewModel(customerId: customer.id))
    }

    var body: some View {
        VStack(spacing: 0) {
            // AI status toggle bar
            HStack {
                Image(systemName: isAIPaused ? "pause.circle.fill" : "cpu.fill")
                    .foregroundColor(isAIPaused ? .orange : .green)
                Text(isAIPaused ? "AI band hai — aap reply kar rahe hain" : "AI active hai")
                    .font(.caption)
                Spacer()
                Toggle("", isOn: $isAIPaused)
                    .labelsHidden()
                    .onChange(of: isAIPaused) { paused in
                        Task { await vm.toggleAI(paused: paused) }
                    }
            }
            .padding(.horizontal)
            .padding(.vertical, 8)
            .background(Color(.systemBackground))
            .overlay(Divider(), alignment: .bottom)

            // Messages ScrollView
            ScrollViewReader { proxy in
                ScrollView {
                    LazyVStack(spacing: 8) {
                        ForEach(vm.messages) { msg in
                            MessageBubble(message: msg)
                                .id(msg.id)
                        }
                    }
                    .padding()
                }
                .onChange(of: vm.messages.count) { _ in
                    if let last = vm.messages.last {
                        proxy.scrollTo(last.id, anchor: .bottom)
                    }
                }
            }

            // Manual reply input (shown always, useful when AI paused)
            HStack(spacing: 10) {
                TextField("Apna jawab likhein...", text: $replyText, axis: .vertical)
                    .textFieldStyle(.roundedBorder)
                    .lineLimit(1...4)

                Button {
                    Task {
                        await vm.sendManualReply(replyText, phone: customer.phone)
                        replyText = ""
                    }
                } label: {
                    Image(systemName: "paperplane.fill")
                        .font(.title3)
                        .foregroundColor(replyText.isEmpty ? .gray : .blue)
                }
                .disabled(replyText.isEmpty)
            }
            .padding()
            .background(Color(.systemBackground))
        }
        .navigationTitle(customer.name)
        .navigationBarTitleDisplayMode(.inline)
        .task { await vm.loadMessages() }
    }
}

import Foundation

struct APIClient {
    static let baseURL = "https://your-app.railway.app/api/v1"
    static let businessId = "your-business-uuid"  // From UserDefaults in prod

    static func getStats() async throws -> DashboardStats {
        let url = URL(string: "\(baseURL)/stats?business_id=\(businessId)")!
        let (data, _) = try await URLSession.shared.data(from: url)
        return try JSONDecoder().decode(DashboardStats.self, from: data)
    }

    static func getCustomers(page: Int = 1) async throws -> [Customer] {
        let url = URL(string: "\(baseURL)/customers?business_id=\(businessId)&page=\(page)")!
        let (data, _) = try await URLSession.shared.data(from: url)
        return try JSONDecoder().decode([Customer].self, from: data)
    }

    static func sendManualReply(phone: String, text: String) async throws {
        var req = URLRequest(url: URL(string: "\(baseURL)/manual-reply")!)
        req.httpMethod = "POST"
        req.setValue("application/json", forHTTPHeaderField: "Content-Type")
        req.httpBody = try JSONEncoder().encode(["phone": phone, "text": text])
        let (_, _) = try await URLSession.shared.data(for: req)
    }
}
